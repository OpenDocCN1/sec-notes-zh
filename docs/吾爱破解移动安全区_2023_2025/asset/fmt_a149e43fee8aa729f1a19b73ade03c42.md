## 加密写入代码

```java
public static void a(Context context) { 
    try {

        String raContent = "okkk" + (System.currentTimeMillis() + 28800000L);

        byte[] keyPart1 = {76, 78, 29, -79, -60, -97, -11, -17};
        byte[] keyPart2 = {66, 23, 45};
        byte[] raKey = new byte[11];
        System.arraycopy(keyPart1, 0, raKey, 0, keyPart1.length);
        System.arraycopy(keyPart2, 0, raKey, keyPart1.length, keyPart2.length);

        b(context, "ra.txt", raContent, raKey);

        long current = System.currentTimeMillis();
        String trContent = String.format("%d-%d-%d-%d",
                                                                                     current,
                                                                                     current + 2000,
                                                                                     current + 3000,
                                                                                     current + 10000
                                                                                     );

        byte[] originalKey = {68, 98, 11, 34, 56};
        byte[] trKey = new byte[originalKey.length + 3];
        System.arraycopy(originalKey, 0, trKey, 0, originalKey.length);
        System.arraycopy(new byte[]{66, 23, 45}, 0, trKey, originalKey.length, 3);

        b(context, "tr.tmp", trContent, trKey);

    } catch (Exception e) {
        e.printStackTrace();
    }
}

private static void b(Context context, String fileName, String content, byte[] key) throws Exception {
    SecretKeySpec keySpec = new SecretKeySpec(key, "RC4");
    Cipher cipher = Cipher.getInstance("RC4");
    cipher.init(Cipher.ENCRYPT_MODE, keySpec);
    byte[] encryptedBytes = cipher.doFinal(content.getBytes("UTF-8"));

    String hexString = bytesToHex(encryptedBytes);

    File file = new File(context.getFilesDir(), fileName);
    try (FileOutputStream fos = new FileOutputStream(file)) {
        fos.write(hexString.getBytes("UTF-8"));
    }
}

private static String bytesToHex(byte[] bytes) {
    StringBuilder sb = new StringBuilder();
    for (byte b : bytes) {
        sb.append(String.format("%02x", b & 0xFF));
    }
    return sb.toString();
}

}
```

在`onCreate`里面直接调用`a()`方法就行

## Pine Hook 拦截

```java
try {
    for (Method m : java.io.File.class.getDeclaredMethods()) {
        if (m.getName().equals("delete") && m.getParameterTypes().length == 0) {
            Pine.hook(m, new MethodHook() {
                @Override
                public void beforeCall(Pine.CallFrame callFrame) throws Throwable {
                    java.io.File file = (java.io.File) callFrame.thisObject;
                    if (file.getName().equals("ra.txt")) {
                        callFrame.setResult(true);
                    }
                }
            });
        }
    }
} catch (Exception e) {
}

try {
    for (Method m : java.io.File.class.getDeclaredMethods()) {
        if (m.getName().equals("deleteOnExit") && m.getParameterTypes().length == 0) {
            Pine.hook(m, new MethodHook() {
                @Override
                public void beforeCall(Pine.CallFrame callFrame) throws Throwable {
                    java.io.File file = (java.io.File) callFrame.thisObject;
                    if (file.getName().equals("ra.txt")) {
                        callFrame.setResult(null);
                    }
                }
            });
        }
    }
} catch (Exception e) {
}
```
