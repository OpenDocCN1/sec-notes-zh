## `tr.tmp` 解密

`tr.tmp` 里面的密钥并不是上面那个，原理和 `ra.txt` 一样，所以没必要再重读一遍，这里就不多做描述，直接放上解密脚本。

```java
import javax.crypto.Cipher;
import javax.crypto.spec.SecretKeySpec;
import java.util.Arrays;
import java.util.HexFormat;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.TimeZone;

public class Main {
    public static void main(String[] args) throws Exception {
        String encryptedHex = "d2818fda49b6c6360d7abc4cffda9902b59947ce3088b1907f0b3e2fbb0bdfeee61784a3cb05f550e2524fd602a5052f2c7080dac99555";

        byte[] originalKey = {68, 98, 11, 34, 56};
        byte[] fullKey = new byte[originalKey.length + 3];
        System.arraycopy(originalKey, 0, fullKey, 0, originalKey.length);
        System.arraycopy(new byte[]{66, 23, 45}, 0, fullKey, originalKey.length, 3);

        byte[] encryptedData = HexFormat.of().parseHex(encryptedHex);
        SecretKeySpec rc4Key = new SecretKeySpec(fullKey, "RC4");
        Cipher cipher = Cipher.getInstance("RC4");
        cipher.init(Cipher.DECRYPT_MODE, rc4Key);
        String decryptedResult = new String(cipher.doFinal(encryptedData));
        System.out.println("最终解密结果: " + decryptedResult);

        String[] timestamps = decryptedResult.split("-");
        SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss.SSS");
        sdf.setTimeZone(TimeZone.getTimeZone("UTC"));

        System.out.println("\n转换为日期:");
        for (String timestamp : timestamps) {
            long millis = Long.parseLong(timestamp);
            Date date = new Date(millis);
            System.out.println(sdf.format(date));
        }
    }
}
```

解密结果为：

```text
最终解密结果: 1754719460120-1754719509607-1754719511534-1754719542243

转换为日期:

2025-08-09 06:04:20.120
2025-08-09 06:05:09.607
2025-08-09 06:05:11.534
2025-08-09 06:05:42.243
```

这里是四个时间戳拼接起来的，可能是对时间有啥校验吧。

## 破解思路

知道了这两个的加解密原理，就可以开始破解了。

### Xposed Hook 方案

考虑到这个软件 `onCreate()` 等生命周期方法都被抽了，所以直接用 `Xposed hook` `Application` 的 `onCreate()`，在启动的时候写入这两个文件，并把加密后的文本写进去。

```java
import de.robv.android.xposed.IXposedHookLoadPackage;
import de.robv.android.xposed.XC_MethodHook;
import de.robv.android.xposed.XposedBridge;
import de.robv.android.xposed.XposedHelpers;
import de.robv.android.xposed.callbacks.XC_LoadPackage;
import android.app.Application;
import android.content.Context;
import javax.crypto.Cipher;
import javax.crypto.spec.SecretKeySpec;
import java.io.File;
import java.io.FileOutputStream;
import java.nio.charset.StandardCharsets;
import java.util.Calendar;
import android.content.Intent;
import android.net.Uri;
import android.view.View;

public class MainHook implements IXposedHookLoadPackage {
    private static String bytesToHex(byte[] bytes) {
        StringBuilder sb = new StringBuilder();
        for (byte b : bytes) {
            sb.append(String.format("%02x", b & 0xFF));
        }
        return sb.toString();
    }

    private String generateRaContent() {
        Calendar calendar = Calendar.getInstance();
        calendar.add(Calendar.HOUR_OF_DAY, 8);
        return "okkk" + calendar.getTimeInMillis();
    }

    private String encryptRa(String input) {
        byte[] key = new byte[]{
            76, 78, 29, -79, -60, -97, -11, -17,
            66, 23, 45
        };
        SecretKeySpec keySpec = new SecretKeySpec(key, "RC4");
        Cipher cipher = Cipher.getInstance("RC4");
        cipher.init(Cipher.ENCRYPT_MODE, keySpec);
        return bytesToHex(cipher.doFinal(input.getBytes(StandardCharsets.UTF_8)));
    }

    private String generateTrContent() {
        long base = System.currentTimeMillis();
        return String.format("%d-%d-%d-%d",
                base,
                base + 2000,
                base + 3000,
                base + 10000);
    }

    private String encryptTr(String input) {
        byte[] key = new byte[]{
            68, 98, 11, 34, 56,
            66, 23, 45
        };
        SecretKeySpec keySpec = new SecretKeySpec(key, "RC4");
        Cipher cipher = Cipher.getInstance("RC4");
        cipher.init(Cipher.ENCRYPT_MODE, keySpec);
        return bytesToHex(cipher.doFinal(input.getBytes(StandardCharsets.UTF_8)));
    }

    @Override
    public void handleLoadPackage(final XC_LoadPackage.LoadPackageParam lpparam) throws Throwable {
        if (!lpparam.packageName.equals("com.layaboxhmhz.gamehmhz.okys")) return;

        XposedHelpers.findAndHookMethod(Application.class, "attach", Context.class, new XC_MethodHook() {
            @Override
            protected void beforeHookedMethod(MethodHookParam param) {
                final Context context = (Context) param.args[0];
                XposedHelpers.findAndHookMethod(Application.class, "onCreate", new XC_MethodHook() {
                    @Override
                    protected void afterHookedMethod(MethodHookParam param) {
                        String raRaw = generateRaContent();
                        String raEncrypted = encryptRa(raRaw);
                        if (raEncrypted != null) {
                            writeFile(context, "ra.txt", raEncrypted, "[RA]");
                        }

                        String trRaw = generateTrContent();
                        String trEncrypted = encryptTr(trRaw);
                        if (trEncrypted != null) {
                            writeFile(context, "tr.tmp", trEncrypted, "[TR]");
                        }
                    }
                });
            }
        });
    }

    private void writeFile(Context context, String filename, String content, String tag) {
        File file = new File(context.getFilesDir(), filename);
        if (file.exists() && !file.delete()) {
            return;
        }
        try (FileOutputStream fos = new FileOutputStream(file)) {
            fos.write(content.getBytes(StandardCharsets.UTF_8));
        }
    }

    private void deleteFiles(Context context) {
        String[] files = {"ra.txt", "tr.tmp"};
        for (String filename : files) {
            File file = new File(context.getFilesDir(), filename);
            if (file.exists() && file.delete()) {
            }
        }
    }
}
```
