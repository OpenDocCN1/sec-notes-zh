XposedHelpers.findAndHookMethod("com.zj.wuaipojie.util.DESUtils", loadPackageParam.classLoader, "des", String.class, new XC_MethodHook() {
            @Override
            protected void beforeHookedMethod(MethodHookParam param) throws Throwable {
                super.beforeHookedMethod(param);
                // 打印原输入参数
                String originalInput = (String) param.args[0];
                XposedBridge.log("[DESUtils.des] 原输入1：" + originalInput);

                param.setResult("220-284");
                XposedBridge.log("[DESUtils.des] 已替换返回值为：220-284");

            }

            @Override
            protected void afterHookedMethod(MethodHookParam param) throws Throwable {
                super.afterHookedMethod(param);
                String aoriginalInput = (String) param.args[0];
                XposedBridge.log("[DESUtils.des] 原输入2：" + aoriginalInput);
            }
        });
