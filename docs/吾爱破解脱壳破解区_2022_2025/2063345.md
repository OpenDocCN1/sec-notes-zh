def extract_map_and_convert(text):
    # 使用正则表达式匹配Map内容
    map_regex = r"Map\(\d+\)\s*\{([^}]+)\}"
    match = re.search(map_regex, text)

    if not match:
        return {}

    map_content = match.group(1)
    lines = [line.strip() for line in map_content.split('\n') if line.strip()]
    result = {}

    for line in lines:
        # 匹配键值对
        key_value_match = re.match(r"\s*'([^']+)' =>\s*(.+)\s*,?\s*$", line)
        if key_value_match:
            key = key_value_match.group(1).lower()
            value = key_value_match.group(2).strip()

            # 去除值两端的单引号（如果存在）
            if value.startswith("'") and value.endswith("'"):
                value = value[1:-1]

            # 去除末尾的逗号（如果存在）
            if value.endswith(','):
                value = value[:-1]

            # 再次检查是否还有单引号（嵌套情况）
            if value.startswith("'") and value.endswith("'"):
                value = value[1:-1]

            # 确保值是字符串
            result[key] = str(value)

    return result
