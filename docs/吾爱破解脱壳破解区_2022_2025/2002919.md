def decrypt():
    # 密文（19字节）
    ciphertext = bytes.fromhex("F75942B63AE26B0C72079872ECF89BAF8F2748")

    # 密钥（19字节）

    key = bytes.fromhex("8A7077840AD040681B72B0478895E0C8EE4B2E")

    # 解密
    plaintext = bytearray()
    for i in range(len(ciphertext)):
        plaintext.append(ciphertext[i] ^ key[i % len(key)])

    return plaintext

result = decrypt()
