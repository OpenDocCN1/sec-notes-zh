```asm
add dword [esi+0x00000110], ebp //还原HOOK的代码
cmp dword [esi+0x00000110], 0x03 判断 小推车数
jc label1  小于3 则跳到过关的call
jnbe label2 否则跳回原代码执行
//以下是过关call
label1: 
pushad 
mov ecx, dword [0x006A9EC0]
mov ecx, dword [ecx+0x00000768]
mov ebx, 0x0040C3E0
call ebx
popad 
//以下是跳回原代码
label2:
mov edx, 0x0041BDEE
jmp edx
```
