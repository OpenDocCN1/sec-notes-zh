function ToHex(src){
	var data = 0;
	len = string.len(src);
	for(i = 1; len; 2){
		one = tonumber('0x' + string.slice(src, i, i + 1,));
		for(j = 1; 8){
			if((((one >> (j - 1)) & 1) ^ (data & 1)) == 1){
				data = ((data ^ 0x18) >> 1) | 0x80;
			}
			else{
				data = data >> 1;
			}
		}
	}
	return string.format("%02X",data);
}
