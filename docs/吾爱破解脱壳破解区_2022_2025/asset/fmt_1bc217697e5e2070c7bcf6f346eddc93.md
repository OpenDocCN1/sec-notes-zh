```

{

  void *v0; // rax

  id v1; // rax

  void ***v2; // rax

  void *v3; // rbx

  void *v4; // rax

  id v5; // r14

  if ( (unsigned __int8)sub_1762768() )         // Is2021

  {

    v0 = (void *)sub_17772D5();

    v1 = objc_retainAutoreleasedReturnValue(v0);

  }

  else
```
