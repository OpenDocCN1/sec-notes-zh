with open("/home/kali/Desktop/re/d3thon/okokokok.txt","r") as f:

    for i in opcodes[::-1]:
        i=i.replace("\'","")
        if "IEKMEDdrPpzpdKy:flag:" in i:
            res -= int(i.split(":")[-1])
        elif "FLNPsiCIvICFtzpUAR:flag:" in i:

            raise SyntaxError
print(hex(res))
