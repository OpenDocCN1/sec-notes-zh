
## 随机字符串生成代码

```js
const randomHelper = {
  cgmkT(x, y) {
    return Math.floor(x * y);
  }
};

function generateRandomString(config) {
  for (var length = config.size, pool = config.num, result = ''; length--; )
      result += pool[randomHelper.cgmkT(Math.random(), pool.length) | 0];
  console.log(config)
  return result;
}
```
