from utils import AESCrypto
import hashlib
import json

def get_md5_string(s: str):
    return hashlib.md5(s.encode('utf-8')).hexdigest()

def decrypt(e: str):
    t = get_md5_string("固定字符串")
    i, r = t[:16], t[16:]
    cipher = AESCrypto(key=r, mode='CBC', padding='Pkcs7')
    return cipher.decrypt(e, iv=i)
