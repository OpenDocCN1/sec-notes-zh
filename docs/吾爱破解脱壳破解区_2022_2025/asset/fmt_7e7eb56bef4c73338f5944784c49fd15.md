## `squareline_lic` 的来源

经过搜索 发现了`UpdateExpireDate`函数内对`squareline_lic`进行了写入 下面就看看`self.squareline_lic`是怎么来的吧 这个倒是没加什么字节码混淆 不知道作者是怎么想的

```python
def UpdateExpireDate(self):
    try:
        self.squareline_lic = ''
        url = f"{self.app_url}/device/status?devicename={self.machine_code}"
        response = requests.get(url)
        response.encoding = 'utf-8'
        if response.status_code == requests.codes.ok:
            Util.myprint('许可证时间请求成功')
            Util.myprint(response.text)
            data_obj = json.loads(response.text)
            status = data_obj['status']
            if status == 'success':
                data = data_obj['data']
                server_time_str = data_obj['time']
                key_str = 'lvgl-gui by fish.2023.889'
                iv_str = 'lvgl-gui by fish.2023.223'
                key_base64 = Util.encode_to_base64(Util.calculate_sha1(key_str))
                iv_base64 = Util.encode_to_base64(Util.calculate_sha1(iv_str))
                decrypt_str = Util.aes_cbc_decrypt_base64(data, key_base64, iv_base64)
                Util.myprint('解密：' + decrypt_str)
                expira_obj = json.loads(decrypt_str)
                expira_date_str = expira_obj['expira_date']
                expira_date = Util.string_to_datetime(expira_date_str)
                local_expira_date_str = Util.datetime_to_string(Util.utc_to_local(expira_date))
                now_date = Util.string_to_datetime(server_time_str)
                Util.set_sys_time(Util.utc_to_local(now_date))
                is_effective = expira_obj['is_effective']
                self.squareline_lic = expira_obj['squareline_lic']
                if is_effective:
                    shouwText = ''
                    self.expira_date = expira_date
                    if expira_date > now_date:
                        self.is_active = True
                        shouwText = '(有效)'
                    else:
                        self.is_active = False
                        shouwText = '(超期)'
                    self.ui.label_expire_date.setText(local_expira_date_str + shouwText)
                else:
                    self.ui.label_expire_date.setText(local_expira_date_str + '(无效)')
                    self.is_active = False
                if self.squareline_lic == '':
                    self.ui.label_expire_date.setText('许可证无效')
                    self.is_active = False
                else:
                    self.is_active = False
                self.ui.label_expire_date.setText('无效')
            else:
                self.is_active = False
                self.ui.label_expire_date.setText('无效')
                Util.myprint('许可证时间请求失败')
                Util.myprint(response.status_code)
        elif not self.is_active:
            self.ui.btn_active_squareline.setEnabled(False)
            self.ui.btn_active_squareline.setStyleSheet('background-color: rgb(203, 203, 203);color: rgb(0, 0, 0);')
            self.ui.label_expire_date.setText('无效')
            software_name = 'SquareLine'
            SquareLine_sq_dat = os.path.join(Util.get_SquareLine_Studio_data_path(software_name), 'sq.dat')
            SquareLine_fdm_dat = os.path.join(Util.get_SquareLine_Studio_data_path(software_name), 'fdm.dat')
            Util.del_file(SquareLine_sq_dat)
            Util.del_file(SquareLine_fdm_dat)
        else:
            self.ui.btn_active_squareline.setEnabled(True)
            self.ui.btn_active_squareline.setStyleSheet('background-color: rgb(157, 105, 0);color: rgb(72, 216, 106);')
            QTimer.singleShot(1000, self.check_dll_new)
    except Exception as e:
        try:
            Util.myprint(e)
            software_name = 'SquareLine'
            SquareLine_sq_dat = os.path.join(Util.get_SquareLine_Studio_data_path(software_name), 'sq.dat')
            SquareLine_fdm_dat = os.path.join(Util.get_SquareLine_Studio_data_path(software_name), 'fdm.dat')
            Util.del_file(SquareLine_sq_dat)
            Util.del_file(SquareLine_fdm_dat)
        finally:
            e = None
            del e
```
