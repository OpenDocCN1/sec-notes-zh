def detect_slide_distance(bg_url: str, front_url: str) -> int:
    """
    使用ddddocr识别滑块缺口位置
    Args:
        bg_url: 背景图URL
        front_url: 滑块图URL
    Returns:
        滑动距离(px)
    """
    # 下载图片
    bg_image = requests.get(bg_url).content
    front_image = requests.get(front_url).content
    det = ddddocr.DdddOcr(det=False, ocr=False,show_ad=False)

    """
