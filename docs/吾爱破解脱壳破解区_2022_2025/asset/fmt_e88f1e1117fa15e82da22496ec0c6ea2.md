
# DLL 加载监控 Hook 程序说明

## 代码实现

```cpp
#include 
#include 
#include 
#include 
#include 
#include 

#pragma comment(lib, "detours.lib")
#pragma comment(lib, "ntdll.lib")

// ================== 数据结构定义 ==================

typedef struct _UNICODE_STRING {
    USHORT Length;
    USHORT MaximumLength;
    PWSTR Buffer;
} UNICODE_STRING, * PUNICODE_STRING;

// ================== 原始函数指针声明 ==================

static HMODULE(WINAPI* TrueLoadLibraryA)(LPCSTR lpLibFileName) = LoadLibraryA;
static HMODULE(WINAPI* TrueLoadLibraryW)(LPCWSTR lpLibFileName) = LoadLibraryW;
static HMODULE(WINAPI* TrueLoadLibraryExA)(LPCSTR lpLibFileName, HANDLE hFile, DWORD dwFlags) = LoadLibraryExA;
static HMODULE(WINAPI* TrueLoadLibraryExW)(LPCWSTR lpLibFileName, HANDLE hFile, DWORD dwFlags) = LoadLibraryExW;

typedef NTSTATUS(NTAPI* PLdrLoadDll)(
    PWCHAR PathToFile,
    ULONG Flags,
    PUNICODE_STRING ModuleFileName,
    PHANDLE ModuleHandle
    );

static PLdrLoadDll TrueLdrLoadDll = nullptr;

// ================== 日志记录函数 ==================

void LogLoadEvent(const char* eventType, const wchar_t* moduleName) {
    wchar_t buffer[512];
    SYSTEMTIME st;
    GetLocalTime(&st);
    // 添加 [HOOKING] 前缀
    swprintf_s(buffer, L"[HOOKING] [%04d-%02d-%02d %02d:%02d:%02d] %hs: %ls",
        st.wYear, st.wMonth, st.wDay,
        st.wHour, st.wMinute, st.wSecond,
        eventType, moduleName);

    OutputDebugStringW(buffer);
}

// ================== Hook 函数实现 ==================

HMODULE WINAPI HookedLoadLibraryA(LPCSTR lpLibFileName) {
    if (lpLibFileName) {
        USES_CONVERSION;
        LogLoadEvent("LoadLibraryA", A2W(lpLibFileName));
    }
    else {
        LogLoadEvent("LoadLibraryA", L"(null)");
    }
    return TrueLoadLibraryA(lpLibFileName);
}

HMODULE WINAPI HookedLoadLibraryW(LPCWSTR lpLibFileName) {
    LogLoadEvent("LoadLibraryW", (lpLibFileName ? lpLibFileName : L"(null)"));
    return TrueLoadLibraryW(lpLibFileName);
}

HMODULE WINAPI HookedLoadLibraryExW(LPCWSTR lpLibFileName, HANDLE hFile, DWORD dwFlags) {
    LogLoadEvent("LoadLibraryExW", (lpLibFileName ? lpLibFileName : L"(null)"));
    return TrueLoadLibraryExW(lpLibFileName, hFile, dwFlags);
}

NTSTATUS NTAPI HookedLdrLoadDll(
    PWCHAR PathToFile,
    ULONG Flags,
    PUNICODE_STRING ModuleFileName,
    PHANDLE ModuleHandle
) {
    LogLoadEvent("LdrLoadDll", (ModuleFileName && ModuleFileName->Buffer) ? ModuleFileName->Buffer : L"(null)");
    return TrueLdrLoadDll(PathToFile, Flags, ModuleFileName, ModuleHandle);
}

// ================== 初始化与 Hook 管理 ==================

void InitLdrLoadDllHook() {
    HMODULE hNtdll = GetModuleHandleW(L"ntdll.dll");
    if (hNtdll) {
        TrueLdrLoadDll = (PLdrLoadDll)GetProcAddress(hNtdll, "LdrLoadDll");
    }
}

void InstallHooks() {
    DetourRestoreAfterWith();
    DetourTransactionBegin();
    DetourUpdateThread(GetCurrentThread());

    DetourAttach(&(PVOID&)TrueLoadLibraryA, HookedLoadLibraryA);
    DetourAttach(&(PVOID&)TrueLoadLibraryW, HookedLoadLibraryW);
    DetourAttach(&(PVOID&)TrueLoadLibraryExW, HookedLoadLibraryExW);

    if (TrueLdrLoadDll) {
        DetourAttach(&(PVOID&)TrueLdrLoadDll, HookedLdrLoadDll);
    }

    LONG error = DetourTransactionCommit();
    if (error != NO_ERROR) {
        LogLoadEvent("Error", L"DetourTransactionCommit failed.");
    }
}

void UninstallHooks() {
    DetourTransactionBegin();
    DetourUpdateThread(GetCurrentThread());

    DetourDetach(&(PVOID&)TrueLoadLibraryA, HookedLoadLibraryA);
    DetourDetach(&(PVOID&)TrueLoadLibraryW, HookedLoadLibraryW);
    DetourDetach(&(PVOID&)TrueLoadLibraryExW, HookedLoadLibraryExW);

    if (TrueLdrLoadDll) {
        DetourDetach(&(PVOID&)TrueLdrLoadDll, HookedLdrLoadDll);
    }

    DetourTransactionCommit();
}

// ================== DLL 入口点 ==================

BOOL APIENTRY DllMain(HMODULE hModule, DWORD ul_reason_for_call, LPVOID lpReserved) {
    switch (ul_reason_for_call) {
    case DLL_PROCESS_ATTACH:
    {
        // 添加测试日志，确认 DLL 成功加载
        OutputDebugStringW(L"[HOOKING] >>>>>>>>>>>> hooking.dll 已注入  DWORD {
            Sleep(300); // 等待进程初始化

            InitLdrLoadDllHook();
            InstallHooks();

            return 0;
            }, nullptr, 0, nullptr);

        if (hThread) {
            CloseHandle(hThread); // 立即关闭句柄，避免资源泄露
        }
    }
    break;

    case DLL_PROCESS_DETACH:
        UninstallHooks();
        break;

    case DLL_THREAD_ATTACH:
    case DLL_THREAD_DETACH:
        break;
    }
    return TRUE;
}
```

## 注入与导出表问题

这个在压缩包 `hooking` 里也有。

代码中 `hooking` 前缀及 `>>>>已注入` 字样是方便之后监听日志时筛选用的。

因为我们编写的 dll 文件之后会需要用到 `detours` 中的 `withdll.exe` 工具注入进程运行，它要求被注入的的 dll 必须导出一个函数且位于导出表的第一个位置。

我们的 dll 没有显示导出函数，会导致报错：

```text
withdll.exe: Error: ...hooking.dll does not export ordinal #1.

See help entry DetourCreateProcessWithDllEx in Detours.chm.
```

代码编写的时候确实有问题，我认罪《^^!》

### 使用 `hooking.def` 解决导出表报错

这里就采用了 `hooking.def` 文件来解决报错（也包含在压缩包的 `hooking` 项目中）。

创建方法就是在项目目录下先创建一个文本文件，往里面写入：

```text
EXPORTS    DllMain @1 PRIVATE
```

然后改后缀为 `.def`，再添加到项目中。
