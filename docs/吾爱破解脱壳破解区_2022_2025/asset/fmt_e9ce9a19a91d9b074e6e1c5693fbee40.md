def jsonp_to_json(jsonp_str):
    """
    将 JSONP 字符串转换为 JSON 对象

    参数:
        jsonp_str (str): JSONP 格式的字符串

    返回:
        dict: 转换后的 JSON 对象
    """
    try:
        # 使用正则表达式提取 JSON 部分
        match = re.search(r'^\w+\(({.*})\)$', jsonp_str)
        if match:
            json_str = match.group(1)
            return json.loads(json_str)
        else:
            # 如果正则匹配失败，尝试直接解析
            return json.loads(jsonp_str)
    except json.JSONDecodeError as e:
        print(f"JSON 解析错误: {e}")
        return None

function rsaEncrypt(data, publicKey) {

    try {

        // 标准化公钥格式

        let formattedKey = publicKey.trim();

        // 如果是单行格式，重新格式化

        if (!formattedKey.includes('\n') && formattedKey.startsWith('-----BEGIN PUBLIC KEY-----')) {

            const keyContent = formattedKey

                .replace('-----BEGIN PUBLIC KEY-----', '')

                .replace('-----END PUBLIC KEY-----', '')

                .replace(/\s/g, '');

            const chunks = keyContent.match(/.{1,64}/g) || [];

            formattedKey = [

                '-----BEGIN PUBLIC KEY-----',

                ...chunks,

                '-----END PUBLIC KEY-----'

            ].join('\n');

        }

        const buffer = Buffer.from(data);

        const encrypted = crypto.publicEncrypt({

            key: formattedKey,

            padding: crypto.constants.RSA_PKCS1_PADDING

        }, buffer);

        return encrypted.toString('base64');

    } catch (error) {

        console.error('RSA encryption failed:', error.message);

        // 返回原始数据作为备选方案

        return Buffer.from(data).toString('base64');

    }
