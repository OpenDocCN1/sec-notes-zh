def sort_dict_by_key(input_dict):
    """字典按照键排序"""
    sorted_keys = sorted(input_dict.keys())
    sorted_dict = {key: input_dict[key] for key in sorted_keys}
    return sorted_dict

def splicing_obj(obj: dict):

        elif isinstance(v, int):
