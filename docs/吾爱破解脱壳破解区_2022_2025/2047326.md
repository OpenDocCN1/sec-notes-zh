(function (){
    'use strict';
    var cookieTemp = '';
    Object.defineProperty(document, 'cookie', {

        set: function (value) {
            if (value.indexOf('v') !== -1) {
                debugger
            }
            return  value;
        },
         get: function () {

            return cookieTemp;
    },
    });
})();
