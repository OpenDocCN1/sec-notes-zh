4> Python分割图片的还原

1、根据坐标及资源id获取imageUrl

```
import hashlib
import time
import math

def get_params(original_url: str):
    a = "xxxx"
    t = hex(math.ceil(math.ceil(time.time() * 1000 / 31536e6) * 31536e3))[2:]
    image_path = original_url.split('/', 3)[-1]
    c = f'{a}/{image_path}{t}'
    sign = hashlib.md5(c.encode()).hexdigest()
    return sign, t

def get_image_url(x, y, resource_id):
    original_url = f"https://xxx/cagstore/{resource_id}/18/{x}_{y}.jpg"
    sign, t = get_params(original_url)
    return f'{original_url}?sign={sign}&t={t}'
