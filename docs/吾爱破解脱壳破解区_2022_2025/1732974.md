```
// ==UserScript==
// @name        六间房免登录看高清
// @namespace   Violentmonkey Scripts
// @match        https://v.6.cn/*
// @grant       none
// @version     1.0
// @AuThor      boxer
// @description 2023/1/4 下午7:31:55
// ==/UserScript==
!(function(){
var tmId = setInterval(function(){
  if(!window.jQuery) return;
  var el = jQuery(".coopPlayerPop-show");
  if(el.length){
    el.remove()
    jQuery(".player-float").css("filter","none");
    clearInterval(tmId)
  }
}, 100)
})()
```
