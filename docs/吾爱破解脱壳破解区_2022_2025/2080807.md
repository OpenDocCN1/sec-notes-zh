def custom_hash_simple(data: bytes) -> int:
    hash_value = 2596069104
    for i, b in enumerate(data):
        hash_value = (hash_value * 2654435761 + b + (i ^ 21930)) & 0xFFFFFFFF
        hash_value = (hash_value ^ (hash_value >> 11)) & 0xFFFFFFFF
